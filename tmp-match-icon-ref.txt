export type MatchIconName = keyof typeof ICONS;

/** Tamanho padrão dos ícones de timeline / disputa. */
export const MATCH_ICON_SIZE = 20;

function tintXml(xml: string, color: string): string {
  return xml
    .replace(/fill="#[0-9A-Fa-f]{3,8}"/g, `fill="${color}"`)
    .replace(/fill='#[0-9A-Fa-f]{3,8}'/g, `fill='${color}'`);
}

type Props = {
  name: MatchIconName;
  size?: number;
  color?: string;
};

/** Ícone de evento — origem `arq/svg`. */
export function MatchIcon({
  name,
  size = MATCH_ICON_SIZE,
  color,
}: Props) {
  const xml = color ? tintXml(ICONS[name], color) : ICONS[name];
  return (
    <View style={[styles.wrap, { width: size, height: size }]}>
      <SvgXml xml={xml} width={size} height={size} />
    </View>
  );
}

const styles = StyleSheet.create({
  wrap: {
    alignItems: 'center',
    justifyContent: 'center',
  },
});
